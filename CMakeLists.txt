cmake_minimum_required(VERSION 3.24)
project(AtlasTaskEngine VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(ATLAS_BUILD_TESTS "Build deterministic and stress tests" ON)
option(ATLAS_SANITIZE_ADDRESS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(ATLAS_SANITIZE_THREAD "Enable ThreadSanitizer" OFF)
if(ATLAS_SANITIZE_ADDRESS AND ATLAS_SANITIZE_THREAD)
  message(FATAL_ERROR "AddressSanitizer and ThreadSanitizer require separate builds")
endif()
find_package(Threads REQUIRED)
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_LIBRARIES Threads::Threads)
check_cxx_source_compiles("#include <thread>
#include <stop_token>
int main() { std::jthread worker([] {}); std::stop_source source; return source.stop_possible() ? 0 : 1; }
" ATLAS_HAS_CXX20_THREADS)
unset(CMAKE_REQUIRED_LIBRARIES)
if(NOT ATLAS_HAS_CXX20_THREADS)
  message(FATAL_ERROR "This compiler AND standard library must support C++20 std::jthread and std::stop_token. Use a recent GCC/libstdc++, MSVC, or Clang/libc++ toolchain.")
endif()

add_library(atlas STATIC src/engine.cpp)
add_library(atlas::atlas ALIAS atlas)
target_include_directories(atlas PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_compile_features(atlas PUBLIC cxx_std_20)
set_target_properties(atlas PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(atlas PUBLIC Threads::Threads)

if(MSVC)
  target_compile_options(atlas PUBLIC /W4 /permissive- /utf-8)
else()
  target_compile_options(atlas PUBLIC -Wall -Wextra -Wpedantic)
endif()
if(ATLAS_SANITIZE_ADDRESS OR ATLAS_SANITIZE_THREAD)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" OR WIN32)
    message(FATAL_ERROR "Sanitizer presets are configured for GCC/Clang on Unix")
  endif()
  if(ATLAS_SANITIZE_ADDRESS)
    set(ATLAS_SANITIZER "-fsanitize=address,undefined")
  else()
    set(ATLAS_SANITIZER "-fsanitize=thread")
  endif()
  target_compile_options(atlas PUBLIC ${ATLAS_SANITIZER} -fno-omit-frame-pointer)
  target_link_options(atlas PUBLIC ${ATLAS_SANITIZER})
endif()

add_executable(atlas-cli app/main.cpp)
target_link_libraries(atlas-cli PRIVATE atlas::atlas)
add_executable(atlas-example examples/usage.cpp)
target_link_libraries(atlas-example PRIVATE atlas::atlas)

if(ATLAS_BUILD_TESTS)
  enable_testing()
  add_executable(atlas-tests tests/engine_tests.cpp)
  target_link_libraries(atlas-tests PRIVATE atlas::atlas)
  add_test(NAME engine.unit COMMAND atlas-tests)
  set_tests_properties(engine.unit PROPERTIES TIMEOUT 60)
  add_test(NAME engine.stress COMMAND atlas-tests --stress)
  set_tests_properties(engine.stress PROPERTIES TIMEOUT 60)
  add_test(NAME cli.workload COMMAND atlas-cli --workers 2 --jobs 8 --limit 1000 --queue 16 --json)
  set_tests_properties(cli.workload PROPERTIES TIMEOUT 20 PASS_REGULAR_EXPRESSION "\"accepted\": 8")
  add_test(NAME cli.invalid-input COMMAND atlas-cli --jobs -1)
  set_tests_properties(cli.invalid-input PROPERTIES WILL_FAIL TRUE TIMEOUT 5)
  add_test(NAME example.usage COMMAND atlas-example)
  set_tests_properties(example.usage PROPERTIES TIMEOUT 20)
endif()
